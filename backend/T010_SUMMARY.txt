SCHEMA CREATED
================================================================================

TABLE: embeddings
├─ summary_id (UUID, PK, FK→summaries)
├─ embedding_vector (pgvector(384), NOT NULL) - SBERT 384D vectors
├─ model_version (VARCHAR, DEFAULT='all-MiniLM-L6-v2')
├─ created_at (TIMESTAMP)
└─ INDEX: idx_embeddings_model_version

TABLE: clusters
├─ cluster_id (UUID, PK)
├─ round_id (UUID, FK→rounds)
├─ user_count (INT, CHECK > 0)
├─ user_pct (FLOAT, CHECK 0 < x ≤ 1.0)
├─ label_summary_id (UUID, FK→summaries) - medoid label
├─ centroid_vector (pgvector(384), NOT NULL) - cluster mean
├─ display_group_id (UUID, nullable) - alignment grouping
├─ created_at (TIMESTAMP)
└─ INDEXES:
   ├─ idx_clusters_round
   ├─ idx_clusters_label_summary
   ├─ idx_clusters_centroid (IVFFlat, cosine_ops)
   └─ idx_clusters_display_group

TABLE: cluster_members
├─ cluster_id (UUID, PK, FK→clusters)
├─ summary_id (UUID, PK, FK→summaries)
├─ user_id (UUID)
├─ CONSTRAINT: UNIQUE(cluster_id, user_id)
└─ INDEXES:
   ├─ idx_cluster_members_summary
   └─ idx_cluster_members_user

TABLE: alignment_maps
├─ alignment_id (UUID, PK)
├─ discussion_id (UUID, FK→discussions)
├─ round_r (INT)
├─ round_r1 (INT) - must be r+1
├─ cluster_r_id (UUID, FK→clusters)
├─ cluster_r1_id (UUID, FK→clusters)
├─ similarity_score (FLOAT, CHECK 0 ≤ x ≤ 1.0)
├─ display_group_id (UUID, nullable)
├─ created_at (TIMESTAMP)
└─ INDEXES:
   ├─ idx_alignment_discussion
   ├─ idx_alignment_display_group
   └─ idx_alignment_clusters
